import java.util.*;
class Main{
    public static void main(String[] args){
        Scanner sc = new Scanner(System.in);
        System.out.print("Input the array: ");
        String in = sc.nextLine();
        String[] n = in.split(" ");
        int p = n.length;
        int arr[] = new int[p];
        ArrayList<Integer> app = new ArrayList<>();
         for(int i = 0;i<p;i++){
             arr[i] = Integer.parseInt(n[i]);
         }
         int k = sc.nextInt();
         for(int i=0; i<=p-k;i++){
            int max = arr[i];
            for(int j=i;j<i+k;j++){
                max = Math.max(max, arr[j]);
            }
            app.add(max);
         }
         System.out.print(app +" ");
    }
}
